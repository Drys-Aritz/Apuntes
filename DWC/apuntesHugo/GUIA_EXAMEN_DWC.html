01Arrays de objetos

Te dan un array de objetos y te piden cuatro funciones: mayúsculas, filtrar por categoría, buscar por título (sin distinguir mayúsculas) y media.

let peliculas = [
  { titulo: "Matrix", director: "Wachowski", anio: 1999, categoria: "Ciencia Ficción", valoracion: 8.7 },
  { titulo: "El Padrino", director: "Coppola", anio: 1972, categoria: "Drama", valoracion: 9.2 },
  { titulo: "Inception", director: "Nolan", anio: 2010, categoria: "Ciencia Ficción", valoracion: 8.8 }
];

// 1. Muestra títulos en mayúsculas por consola (no devuelve nada)
function mayusculas() {
  for (let i = 0; i < peliculas.length; i++) {
    console.log(peliculas[i].titulo.toUpperCase());
  }
}

// 2. Devuelve un array nuevo con las películas de esa categoría
function peliculasPorCategoria(categoria) {
  let resultado = [];
  for (let i = 0; i < peliculas.length; i++) {
    if (peliculas[i].categoria === categoria) {
      resultado.push(peliculas[i]);
    }
  }
  return resultado;   // [] si no hay ninguna
}

// 3. Devuelve películas cuyo título contenga el texto (sin distinguir mayúsculas)
function buscarPorTitulo(texto) {
  let resultado = [];
  for (let i = 0; i < peliculas.length; i++) {
    if (peliculas[i].titulo.toLowerCase().includes(texto.toLowerCase())) {
      resultado.push(peliculas[i]);   // la película, NO el título
    }
  }
  return resultado;
}

// 4. Devuelve la media de las valoraciones
function mediaValoraciones() {
  let suma = 0;
  for (let i = 0; i < peliculas.length; i++) {
    suma += peliculas[i].valoracion;
  }
  return suma / peliculas.length;
}

// console.log(mediaValoraciones());

Errores que restan puntos: devolver un string tipo "No hay películas" en vez de []; usar console.log dentro de una función que debe devolver; hacer resultado.push(p.titulo) en vez de resultado.push(p).

02Objetos y clases

Te piden una clase con constructor, un método mostrar() que devuelve una frase exacta, dos objetos, un array y recorrerlo por consola.

class Pelicula {
  constructor(titulo, director, anio, categoria, valoracion) {
    this.titulo = titulo;
    this.director = director;
    this.anio = anio;
    this.categoria = categoria;
    this.valoracion = valoracion;
  }

  mostrar() {
    return `${this.titulo} (${this.anio}), dirigida por ${this.director}, en la categoría de ${this.categoria}`;
  }
}

// Dos objetos
let p1 = new Pelicula("Matrix", "Wachowski", 1999, "Ciencia Ficción", 8.7);
let p2 = new Pelicula("El Padrino", "Coppola", 1972, "Drama", 9.2);

let coleccion = [p1, p2];

for (let i = 0; i < coleccion.length; i++) {
  console.log(coleccion[i].mostrar());
}

Getters y setters: si el enunciado pide get/set estándares, usa get color() { return this._color; } y set color(v) { this._color = v; }. Se usan sin paréntesis: d.color.

03DOM y eventos

Botón que recorre el array y crea <li> con createElement + textContent + appendChild (nunca innerHTML). Efecto hover: negrita al entrar, normal al salir.

<!-- En el HTML -->
<script src="dom.js" defer></script>
<button id="btnMostrar">Mostrar</button>
<ul id="lista"></ul>

const boton = document.getElementById("btnMostrar");
const ul = document.getElementById("lista");

boton.addEventListener("click", function () {
  while (ul.firstChild) ul.removeChild(ul.firstChild);   // no duplicar

  for (let i = 0; i < peliculas.length; i++) {
    let li = document.createElement("li");
    li.textContent = peliculas[i].titulo;

    li.addEventListener("mouseover", function () {
      li.style.fontWeight = "bold";
    });
    li.addEventListener("mouseout", function () {
      li.style.fontWeight = "normal";
    });

    ul.appendChild(li);
  }
});

Ojo: el <script> necesita defer o ir al final del <body>, o todo da null. Para saber qué li recibió el evento usa e.target, no querySelector("li") (siempre coge el primero).

04Validación de formularios

Escuchar el submit, e.preventDefault() siempre, validar con regex y escribir en un div.

const form = document.getElementById("formPelicula");
const mensaje = document.getElementById("mensaje");

const reTexto = /^[a-záéíóúñü\s]{3,50}$/i;   // letras, acentos y espacios, 3-50
const reAnio  = /^(19|20)\d{2}$/;             // 1900-2099

form.addEventListener("submit", function (e) {
  e.preventDefault();

  let titulo   = document.getElementById("titulo").value.trim();
  let director = document.getElementById("director").value.trim();
  let anio     = document.getElementById("anio").value.trim();
  let categoria = document.getElementById("categoria").value;

  let errores = [];
  if (!reTexto.test(titulo))   errores.push("Título: 3-50 letras.");
  if (!reTexto.test(director)) errores.push("Director: 3-50 letras.");
  if (!reAnio.test(anio))      errores.push("Año: 1900-2099.");
  if (categoria === "")        errores.push("Elige categoría.");

  if (errores.length > 0) {
    mensaje.textContent = errores.join(" · ");
    mensaje.style.color = "red";
  } else {
    mensaje.textContent = "Película validada correctamente";
    mensaje.style.color = "green";
  }
});

Regex lista para copiar:

  • Texto 3-50 letras (con acentos y espacios): /^[a-záéíóúñü\s]{3,50}$/i
  • Año 1900-2099 (4 cifras): /^(19|20)\d{2}$/

05Web Storage (localStorage)

Guardar un nombre, mostrar "Bienvenido de nuevo, X", ocultar el formulario, botón de cerrar sesión.

const form = document.getElementById("formUsuario");
const bienvenida = document.getElementById("bienvenida");
const cerrar = document.getElementById("cerrarSesion");

function pintar() {
  const usuario = localStorage.getItem("usuario");

  if (usuario) {
    form.style.display = "none";
    bienvenida.textContent = "Bienvenido de nuevo, " + usuario;
    cerrar.hidden = false;
  } else {
    form.style.display = "block";
    bienvenida.textContent = "";
    cerrar.hidden = true;
  }
}

form.addEventListener("submit", function (e) {
  e.preventDefault();
  const usuario = document.getElementById("nombreUsuario").value.trim();
  if (usuario !== "") {
    localStorage.setItem("usuario", usuario);
    pintar();
  }
});

cerrar.addEventListener("click", function () {
  localStorage.removeItem("usuario");
  pintar();
});

pintar();   // al cargar

Cuatro instrucciones que vale saber:

  • localStorage.setItem("clave", valor) — guardar
  • localStorage.getItem("clave") — leer (devuelve null si no existe)
  • localStorage.removeItem("clave") — borrar una
  • JSON.stringify(obj) y JSON.parse(str) — para guardar objetos o arrays
